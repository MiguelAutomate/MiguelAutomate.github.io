ABOUT vortexVote.js

This demo is a browser-side mirror of a small, dependency-free Bun/ES module. It exposes pure functions for each path plus a dispatcher, so the same logic can run in a UI, a CI gate, or a CLI without modification.

import { runVote, VotePath, isOk } from "./vortexVote.js";

const result = runVote(VotePath.MIDDLE, {
  seedVotes:      [{vote:"accept"}, {vote:"accept"}, {vote:"accept"}],
  validatorVotes: [{vote:"accept"}, {vote:"accept"}, {vote:"accept"},
                    {vote:"accept"}, {vote:"reject"}, {vote:"reject"}],
});

if (isOk(result)) {
  result.value; // { outcome: "COMMIT", details: { passes: 9, rounds: [...], passThreshold: 7 } }
} else {
  result.error; // VortexVoteError — malformed input, not a business outcome
}
